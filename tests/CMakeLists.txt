find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(ggpo_peer peer.cpp)
target_link_libraries(ggpo_peer PRIVATE GGPO)
set_target_properties(ggpo_peer PROPERTIES CXX_STANDARD 11)
add_executable(ggpo_checks checks.cpp)
target_link_libraries(ggpo_checks PRIVATE GGPO)
target_include_directories(ggpo_checks PRIVATE ${PROJECT_SOURCE_DIR}/src/lib/ggpo)
set_target_properties(ggpo_checks PROPERTIES CXX_STANDARD 11)
add_executable(ggpo_c_header c_header.c)
target_link_libraries(ggpo_c_header PRIVATE GGPO)
set_target_properties(ggpo_c_header PROPERTIES LINKER_LANGUAGE CXX)
add_test(NAME public_c_header COMMAND ggpo_c_header)
add_test(NAME platform_and_packets COMMAND ggpo_checks)
set_tests_properties(public_c_header platform_and_packets PROPERTIES TIMEOUT 15)
foreach(scenario baseline delayed impaired recovery disconnect disconnect_startup spectator synctest)
    add_test(NAME ${scenario} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/run_peers.py $<TARGET_FILE:ggpo_peer> ${scenario}
        ${CMAKE_CURRENT_BINARY_DIR}/results/${scenario})
    set_tests_properties(${scenario} PROPERTIES TIMEOUT 60)
endforeach()
